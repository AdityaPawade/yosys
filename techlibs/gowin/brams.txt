ram block $__GOWIN_SP_ {
	abits 14;
	widths 9 18 36 per_port;
	cost 128;
	byte 9;
	init no_undef;
	port srsw "A" {
		clock posedge;
		clken;
		option "RESET_MODE" "SYNC" {
			rdsrst zero ungated;
		}
		option "RESET_MODE" "ASYNC" {
			rdarst zero;
		}
		rdinit zero;
		portoption "WRITE_MODE" 0 {
			rdwr no_change;
		}
		portoption "WRITE_MODE" 1 {
			rdwr new;
		}
		ifndef gw5a {
			portoption "WRITE_MODE" 2 {
				rdwr old;
			}
		}
		wrbe_separate;
	}
}

ram block $__GOWIN_SDP_ {
	abits 14;
	widths 1 2 4 8 16 32 per_port;
	# 2026-05-12 Phase 11 — Gowin EDA bit-stripe proof point: decrypted Gowin
	# .vg netlist shows GowinSynthesis maps multi-port sector_buffer to SDPB
	# (8-bit family widths 1/2/4/8/16/32), NOT SDPX9B (9-bit family 1/2/4/9/18/36).
	# Specifically uses 1-bit-write x 32-bit-read SDPB cells in a bit-stripe
	# pattern (8 cells share one logical storage, each handling 1 bit of write
	# data, all serving 32-bit parallel reads at different addresses).
	#
	# Previous patch (2026-05-10, commit 49338eb5) lowered SDP cost to 64 to
	# bias libmap away from DP, but kept widths 9/18/36 -> libmap kept requesting
	# 9-bit widths -> brams_map_gw5a.v fell through to SDPX9B branch.
	# This change forces 8-bit family widths so brams_map_gw5a.v takes the
	# PORT_W_WIDTH < 9 branch and emits SDPB. The SDPB code path was already
	# present (and unreachable) in brams_map_gw5a.v line 348-378 — we just
	# need libmap to request widths < 9 to activate it.
	#
	# byte=8 means write-byte-enable granularity is 8 bits per byte (not 9).
	# True dual-port and 9-bit-family-only memories (rare) fall through to
	# $__GOWIN_DP_ (still has 9/18 in its widths list).
	cost 64;
	byte 8;
	init no_undef;
	port sr "R" {
		clock posedge;
		clken;
		# R22: removed SYNC option to force RESET_MODE=ASYNC, matching V3_GOOD.
		# User wants every SDPB parameter to match working V3 build before
		# concluding investigation.
		option "RESET_MODE" "ASYNC" {
			rdarst zero;
		}
		rdinit zero;
	}
	port sw "W" {
		clock posedge;
		clken;
		wrbe_separate;
	}
}
